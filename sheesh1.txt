const assert = require('assert');


  Simulates a login system.
 
class AuthService {
    constructor() {
         Mock database
        this.validUser = rregual@uspf.edu.ph;
        this.validPass = secret123;
    }

    
      Authenticates a user.
      @param {string} username 
      @param {string} password 
      @returns {object} Result object with success status and message
     
    login(username, password) {
        if (!username  !password) {
            return { success false, message Username and password are required. };
        }

        if (username === this.validUser && password === this.validPass) {
            return { success true, message Login successful! };
        } else if (username !== this.validUser) {
            return { success false, message Wrong username. };
        } else {
            return { success false, message Wrong password. };
        }
    }
}
